```java
/**
 * Jackson 全局序列化配置。
 *
 * 将 Long/long 统一序列化为字符串：雪花 ID 为 19 位数字，超出 JS
 * Number.MAX_SAFE_INTEGER（2^53-1），以 JSON number 下发会导致前端精度丢失。
 * 影响范围见 docs/backed/2026-08-24-后端功能调整.md R0（id/total/各统计计数）。
 * Integer 与浮点（rating/score/price）不受影响。
 */
@Configuration
public class JacksonConfig {

    /**
     * 注册 Long→ToStringSerializer 的全局 ObjectMapper 定制。
     *
     * @param builder Spring Boot 注入的构建器（保留其余默认定制）
     * @return 定制后的 ObjectMapper，由容器接管
     */
    @Bean
    public ObjectMapper objectMapper(Jackson2ObjectMapperBuilder builder) {
        SimpleModule module = new SimpleModule();
        module.addSerializer(Long.class, ToStringSerializer.instance);
        module.addSerializer(Long.TYPE, ToStringSerializer.instance);
        return builder.modules(module).build();
    }
}
```
